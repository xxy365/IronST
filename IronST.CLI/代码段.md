## 查看IR
```
using IronST.Frontend;
using IronST.LLVMCodeGen;

var source = @"
(* 简化版：只测结构体字段读写，不调函数 *)

TYPE Point : STRUCT
    x : INT;
    y : INT;
END_STRUCT; END_TYPE;

TYPE Rect : STRUCT
    left   : INT;
    top    : INT;
    right  : INT;
    bottom : INT;
END_STRUCT; END_TYPE;

TYPE Sensor : STRUCT
    id       : INT;
    value    : REAL;
    active   : BOOL;
END_STRUCT; END_TYPE;

PROGRAM Main
    VAR
        p1       : Point;
        p2       : Point;
        box      : Rect;
        sensor1  : Sensor;
        tmp      : INT;
        tmpR     : REAL;
    END_VAR

    (* Point *)
    p1.x := 10;
    p1.y := 20;
    p2.x := p1.x + p1.y;
    p2.y := p1.x * 2;

    (* Rect 四字段 *)
    box.left   := 0;
    box.top    := 0;
    box.right  := 100;
    box.bottom := 50;

    (* Sensor 混合类型 *)
    sensor1.id     := 7;
    sensor1.value  := 3.14;
    sensor1.active := TRUE;

    (* 读取字段再计算 *)
    tmp  := box.right - box.left;
    tmpR := sensor1.value + 1.0;
END_PROGRAM
";
var pipeline = new FrontendPipeline();
var analysis = FrontendPipeline.Analyze(source);

if (!analysis.Success)
{
    string errors = string.Join("\n", analysis.Diagnostics.Select(d => $"  {d}"));
    throw new Exception(errors);
}

using var irGen = new IRGenerator();
string ir = irGen.Generate(analysis);

Console.WriteLine(ir);
```
